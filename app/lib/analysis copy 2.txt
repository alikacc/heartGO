import { useState, useEffect, useRef } from 'react';
import coeffData from './firCoefficients.json';
const koefFIRmatlab = coeffData.koefFIRmatlab;
import useBLEECG from './landing';

function applyFIRFilter(signal, coeffs) {
    const N = coeffs.length;
    const filtered = new Array(signal.length).fill(0);
    for (let i = 0; i < signal.length; i++) {
        let acc = 0;
        for (let j = 0; j < N; j++) {
            if (i - j >= 0) {
                acc += coeffs[j] * signal[i - j];
            }
        }
        filtered[i] = acc;
    }
    return filtered;
}

function forwardBackwardFIRFilter(signal, coeffs) {
    // Forward pass
    const forward = applyFIRFilter(signal, coeffs);
    // Reverse signal
    const reversed = forward.slice().reverse();
    // Backward filter
    const backward = applyFIRFilter(reversed, coeffs);
    // Reverse again to restore original order
    return backward.reverse();
}

function gradient(array) {
    let grad = new Array(array.length).fill(0.0);
    for (let i = 1; i < array.length - 1; i++) {
        grad[i] = Math.abs((array[i + 1] - array[i - 1]) / 2);
    }
    grad[0] = Math.abs(array[1] - array[0]);
    grad[array.length - 1] = Math.abs(array[array.length - 1] - array[array.length - 2]);
    return grad;
}

function mv_avg(signal, window_size) {
    let half_window = Math.floor(window_size / 2);
    let averaged = [];
    for (let i = 0; i < signal.length; i++) {
        let acc = 0.0;
        for (let j = i - half_window; j <= i + half_window; j++) {
            if (j >= 0 && j < signal.length) {
                acc += signal[j];
            } else {
                acc += 0.0; // Zero-padding
            }
        }
        averaged.push(acc / window_size);
    }
    return averaged;
}

function index_of_max(arr) {
    if (arr.length === 0) return -1;
    let max_index = 0;
    for (let i = 1; i < arr.length; i++) {
        if (arr[i] > arr[max_index]) {
            max_index = i;
        }
    }
    return max_index;
}

function peakRdetect(signal, smoothwindow = 0.1, avgwindow = 0.75,
    gradthreshweight = 1.5, minlenweight = 0.4, mindelay = 0.3) {
    const SAMPLING_RATE = 320;
    let grad = gradient(signal);
    let smthgrd = mv_avg(grad, Math.round(smoothwindow * SAMPLING_RATE));
    let avggrd = mv_avg(smthgrd, Math.round(avgwindow * SAMPLING_RATE));
    let threshgrd = avggrd.map(val => val * gradthreshweight);
    let minDelay = Math.round(mindelay * SAMPLING_RATE);
    let start = [], end = [], duration = [];
    let count = 0, mulai = 0;
    for (let i = 0; i < avggrd.length; i++) {
        if (smthgrd[i] > threshgrd[i] && mulai === 1) {
            count += 1;
        } else if (smthgrd[i] > threshgrd[i] && mulai === 0) {
            start.push(i);
            count += 1;
            mulai = 1;
        } else if (!(smthgrd[i] > threshgrd[i]) && mulai === 1) {
            end.push(i - 1);
            duration.push(count);
            count = 0;
            mulai = 0;
        }
    }
    const mean = arr => arr.reduce((a, b) => a + b, 0) / arr.length;
    let mindur = mean(duration) * minlenweight;
    let peak = [0];
    for (let i = 0; i < duration.length; i++) {
        if (duration[i] < mindur) continue;
        let data = signal.slice(start[i], end[i] + 1);
        let max = index_of_max(data) + start[i];
        if (max - peak[peak.length - 1] > minDelay) {
            peak.push(max);
        }
    }
    peak.shift(); // Remove the initial placeholder 0
    return peak;
}

// Function to calculate 12-lead ECG from Lead I and Lead II
function calculateAllLeads(lead1, lead2) {
    // Ensure both leads have the same length
    const minLength = Math.min(lead1.length, lead2.length);
    const leadI = lead1.slice(0, minLength);
    const leadII = lead2.slice(0, minLength);

    // Calculate Lead III: Lead III = Lead II - Lead I
    const leadIII = leadII.map((val, idx) => val - leadI[idx]);

    // Calculate augmented leads
    // aVR = -(Lead I + Lead II) / 2
    const aVR = leadI.map((val, idx) => -((val + leadII[idx]) / 2));

    // aVL = Lead I - Lead III / 2
    const aVL = leadI.map((val, idx) => val - (leadIII[idx] / 2));

    // aVF = Lead II + Lead III / 2
    const aVF = leadII.map((val, idx) => val + (leadIII[idx] / 2));

    return {
        leadI,
        leadII,
        leadIII,
        aVR,
        aVL,
        aVF
    };
}

// UPDATED: Modified function signature to accept shouldStop parameter
export default function useECGAnalysis(deviceId, shouldStop = false) {
    // Get raw samples from BLE (now returns both leads) - PASS shouldStop parameter
    const bleData = useBLEECG(deviceId, shouldStop);
    const rawLead1 = bleData?.lead1 || [];
    const rawLead2 = bleData?.lead2 || [];

    // State for processed data - now includes all 6 leads
    const [filteredLead1, setFilteredLead1] = useState([]);
    const [filteredLead2, setFilteredLead2] = useState([]);
    const [filteredLead3, setFilteredLead3] = useState([]);
    const [filteredAVR, setFilteredAVR] = useState([]);
    const [filteredAVL, setFilteredAVL] = useState([]);
    const [filteredAVF, setFilteredAVF] = useState([]);

    // Peak detection for all leads
    const [peaksLead1, setPeaksLead1] = useState([]);
    const [peaksLead2, setPeaksLead2] = useState([]);
    const [peaksLead3, setPeaksLead3] = useState([]);
    const [peaksAVR, setPeaksAVR] = useState([]);
    const [peaksAVL, setPeaksAVL] = useState([]);
    const [peaksAVF, setPeaksAVF] = useState([]);

    // Apply filtering and calculate all leads when raw data changes
    useEffect(() => {
        // UPDATED: Clear all data immediately when shouldStop is true
        if (shouldStop) {
            setFilteredLead1([]);
            setFilteredLead2([]);
            setFilteredLead3([]);
            setFilteredAVR([]);
            setFilteredAVL([]);
            setFilteredAVF([]);
            setPeaksLead1([]);
            setPeaksLead2([]);
            setPeaksLead3([]);
            setPeaksAVR([]);
            setPeaksAVL([]);
            setPeaksAVF([]);
            return;
        }

        if (rawLead1.length === 0 || rawLead2.length === 0) {
            // Reset all states if we don't have both leads
            setFilteredLead1([]);
            setFilteredLead2([]);
            setFilteredLead3([]);
            setFilteredAVR([]);
            setFilteredAVL([]);
            setFilteredAVF([]);
            setPeaksLead1([]);
            setPeaksLead2([]);
            setPeaksLead3([]);
            setPeaksAVR([]);
            setPeaksAVL([]);
            setPeaksAVF([]);
            return;
        }

        try {
            // Apply forward-backward FIR filter to raw leads
            const filteredRawLead1 = applyFIRFilter(rawLead1, koefFIRmatlab);
            const filteredRawLead2 = applyFIRFilter(rawLead2, koefFIRmatlab);

            // Calculate all 6 leads from the filtered raw data
            const allLeads = calculateAllLeads(filteredRawLead1, filteredRawLead2);

            // Set filtered data for all leads
            setFilteredLead1(allLeads.leadI);
            setFilteredLead2(allLeads.leadII);
            setFilteredLead3(allLeads.leadIII);
            setFilteredAVR(allLeads.aVR);
            setFilteredAVL(allLeads.aVL);
            setFilteredAVF(allLeads.aVF);

            // Apply peak detection to all leads
            try {
                setPeaksLead1(peakRdetect(allLeads.leadI));
            } catch (e) {
                console.warn('Peak detection failed for Lead I:', e);
                setPeaksLead1([]);
            }

            try {
                setPeaksLead2(peakRdetect(allLeads.leadII));
            } catch (e) {
                console.warn('Peak detection failed for Lead II:', e);
                setPeaksLead2([]);
            }

            try {
                setPeaksLead3(peakRdetect(allLeads.leadIII));
            } catch (e) {
                console.warn('Peak detection failed for Lead III:', e);
                setPeaksLead3([]);
            }

            try {
                setPeaksAVR(peakRdetect(allLeads.aVR));
            } catch (e) {
                console.warn('Peak detection failed for aVR:', e);
                setPeaksAVR([]);
            }

            try {
                setPeaksAVL(peakRdetect(allLeads.aVL));
            } catch (e) {
                console.warn('Peak detection failed for aVL:', e);
                setPeaksAVL([]);
            }

            try {
                setPeaksAVF(peakRdetect(allLeads.aVF));
            } catch (e) {
                console.warn('Peak detection failed for aVF:', e);
                setPeaksAVF([]);
            }

        } catch (error) {
            console.warn('ECG analysis failed:', error);
            // Fallback to raw samples if filtering fails
            setFilteredLead1(rawLead1);
            setFilteredLead2(rawLead2);
            setFilteredLead3([]);
            setFilteredAVR([]);
            setFilteredAVL([]);
            setFilteredAVF([]);
            setPeaksLead1([]);
            setPeaksLead2([]);
            setPeaksLead3([]);
            setPeaksAVR([]);
            setPeaksAVL([]);
            setPeaksAVF([]);
        }
    }, [rawLead1, rawLead2, shouldStop]); // UPDATED: Added shouldStop to dependencies

    return {
        // Raw data
        rawLead1: rawLead1 || [],
        rawLead2: rawLead2 || [],

        // Filtered data for all 6 leads
        filteredLead1: filteredLead1 || [],
        filteredLead2: filteredLead2 || [],
        filteredLead3: filteredLead3 || [],
        filteredAVR: filteredAVR || [],
        filteredAVL: filteredAVL || [],
        filteredAVF: filteredAVF || [],

        // Peak detection results for all leads
        peaksLead1: peaksLead1 || [],
        peaksLead2: peaksLead2 || [],
        peaksLead3: peaksLead3 || [],
        peaksAVR: peaksAVR || [],
        peaksAVL: peaksAVL || [],
        peaksAVF: peaksAVF || [],

        // Processing status
        isProcessing: (rawLead1.length > 0 || rawLead2.length > 0) &&
            (filteredLead1.length === 0 || filteredLead2.length === 0)
    };
}
